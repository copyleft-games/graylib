#define MY_TYPE_APPLICATION (my_application_get_type ())
G_DECLARE_FINAL_TYPE (MyApplication, my_application, MY, APPLICATION, GrlApplication)

struct _MyApplication
{
    GrlApplication parent_instance;
    GrlWindow *window;
};

G_DEFINE_TYPE (MyApplication, my_application, GRL_TYPE_APPLICATION)

static void
my_application_activate (GrlApplication *app)
{
    MyApplication *self = MY_APPLICATION (app);

    self->window = grl_window_new (800, 600, "My Game");
    grl_window_run (self->window);
}

static void
my_application_class_init (MyApplicationClass *klass)
{
    GrlApplicationClass *app_class = GRL_APPLICATION_CLASS (klass);
    app_class->activate = my_application_activate;
}

*** Frames in Flight (Input Latency)
:PROPERTIES:
:CUSTOM_ID: frames-in-flight
:END:
A GL driver lets the CPU queue finished frames ahead of the GPU before it blocks (Mesa about one, other drivers up to three). Every queued frame is input latency: the input a frame was built from is that much older when it reaches the screen. *Frame sync* (=src/core/grl-frame-sync.h=) bounds the queue with GL sync objects: before each buffer swap it fences the frame (=glFenceSync()=) and waits (=glClientWaitSync()=) until at most N - 1 earlier frames are unfinished on the GPU, so with the frame the CPU builds next at most N are in flight.

#+begin_src C
/* on a window: grl_window_end_drawing() and grl_window_swap_buffers() honour it */
grl_window_set_max_frames_in_flight (window, 2);
g_object_set (window, "max-frames-in-flight", 1, NULL);

/* raw raylib loops: grl_frame_sync_end_drawing() in place of EndDrawing() */
grl_frame_sync_set_max_frames_in_flight (2);
BeginDrawing ();
/* ... */
grl_frame_sync_end_drawing ();

gdouble waited = grl_frame_sync_get_last_wait ();   /* seconds the last swap waited on the GPU */
#+end_src

| Bound | Behaviour                                                                                          |
|-------+----------------------------------------------------------------------------------------------------|
| 0     | Off (default): plain =EndDrawing()=, the driver's own limit                                        |
| 1     | The GPU finishes each frame before its swap: the next input poll sees an idle GPU. Lowest latency; CPU and GPU work no longer overlap, so a GPU-bound program loses frame rate |
| 2     | One frame of CPU/GPU overlap (other APIs' "max frame latency 1"). What Mesa does by itself; caps drivers that queue three |
| 3-4   | Looser caps (=GRL_FRAME_SYNC_MAX_FRAMES= is 4)                                                     |

Where the wait sits: =grl_frame_sync_end_drawing()= flushes raylib's render batch, fences and waits, then calls =EndDrawing()= (swap, frame limiter, input poll), so the input the next frame reads is always polled after the wait. raylib's own frame-time accounting (=GetFrameTime()=, =GetFPS()=) is untouched. A fence the GPU already passed is checked without a flush or a block, so on a GPU that keeps up the cost is one fence per frame (measured: no frame-rate change at ~1000 fps on an Iris Xe, a 2 µs median wait on llvmpipe). With the bound at 0 the present path is exactly =EndDrawing()=.

Support: sync objects are core in OpenGL 3.2+ and ES 3.0, so raylib's desktop GL 3.3 build always has them; on GL 1.1/2.1/ES 2.0 or the software renderer =grl_frame_sync_is_supported()= is =FALSE= and every call is a no-op. A wait gives up after 100 ms (counted by =grl_frame_sync_get_timeouts()=) so a lost GPU cannot hang the frame; a =GL_WAIT_FAILED= turns frame sync off for that context with a warning. The bound is process-wide, like raylib's window and GL context; =GrlWindow= deletes the outstanding fences when it closes the window (=grl_frame_sync_reset()=; raw raylib code calls it before =CloseWindow()=).

When to use it: measured on Mesa (Iris Xe, Wayland and XWayland), the driver already holds GPU-bound and vsynced frames to about one frame of queue, so 2 changes nothing there and is a cheap guard elsewhere; 1 cut input-to-GPU-done latency from 3.5 to 2.5 ms on an uncapped GPU-bound scene at a 19% frame-rate cost, and changed nothing with vsync on. libregnum's =docs/performance.org= ("Input latency") has the full tables.

*** Running the Window Loop
:PROPERTIES:
:CUSTOM_ID: running-the-window-loop
:END:
#+begin_src C
/* Option 1: Built-in loop (uses signals) */
grl_window_run (window);

/* Option 2: Manual loop */
while (!grl_window_should_close (window))
{
    grl_window_begin_drawing (window);
    grl_window_clear_background (window, bg_color);
    /* Draw content */
    grl_window_end_drawing (window);
}

#define MY_TYPE_WINDOW (my_window_get_type ())

}

my_window_update (GrlWindow *window,
                  gfloat     delta_time)
{
    /* Update game state */
}

my_window_class_init (MyWindowClass *klass)

    GrlWindowClass *window_class = GRL_WINDOW_CLASS (klass);
    window_class->draw = my_window_draw;
    window_class->update = my_window_update;
}
